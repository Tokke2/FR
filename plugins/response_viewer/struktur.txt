│   │                                          # Fältnamn/Position/Längd/Typ/Obligatorisk/Beskrivning

│   │                                          # _refresh_overview(),
│   │                                          # _refresh_post_types(),
│   │                                          # _refresh_versions()
│   │                                          # _refresh_overview():

│   │                                          # uppdaterar labels,
│   │                                          # räknar total_fields,
│   │                                          # fyller post_table med PTxx, namn, fältantal, total_length
│   │                                          # _refresh_post_types():
│   │                                          # fyller dropdown med "PT{pt} - {name}"
│   │                                          # _refresh_versions():
│   │                                          # uppdaterar versionslista,
│   │                                          # markerar aktiv version med "→",
│   │                                          # fyller compare-dropdowns,
│   │                                          # väljer olika versioner om minst två finns
│   │                                          # _on_post_type_changed(text):
│   │                                          # läser currentData() som posttyp,
│   │                                          # hämtar schema,
│   │                                          # sorterar fields på position,
│   │                                          # fyller fields_table
│   │                                          # _do_compare():
│   │                                          # hämtar v1/v2,
│   │                                          # varnar om samma version,
│   │                                          # använder viewer.compare_versions() eller manager fallback,
│   │                                          # skriver textbaserat resultat:
│   │                                          # added(+ PTxx), removed(- PTxx), modified(~ PTxx)
│   │                                          # _export_json():
│   │                                          # save dialog,
│   │                                          # exporterar aktiv version + alla posttyper/fält till JSON:
│   │                                          # {
│   │                                          #   version,
│   │                                          #   exported_at,
│   │                                          #   post_types: {
│   │                                          #     pt: {
│   │                                          #       name,
│   │                                          #       total_length,
│   │                                          #       fields:[...]
│   │                                          #     }
│   │                                          #   }
│   │                                          # }
│   │                                          # visar QMessageBox vid lyckad/misslyckad export

│   │                                          # visa posttyper,
│   │                                          # visa detaljerade fält per posttyp,
│   │                                          # jämföra två versioner,
│   │                                          # exportera schemaöversikt till JSON
│   │                                          # Begränsningar:
│   │                                          # pluginen är viewer/inspector,
│   │                                          # inte source of truth,
│   │                                          # ändrar inte schema,
│   │                                          # importerar inte schema själv,
│   │                                          # fallbackdiff på modifierade posttyper är relativt grov
